```python
import serial
import math

PORT = "/dev/ttyUSB0"
BAUD = 115200

# 临时基站间距（米）
d23 = 0.31
d24 = 0.70
d34 = 0.80

# 建立临时坐标系
# A2 = (0, 0)
# A3 = (0.31, 0)
x4 = (d24**2 + d23**2 - d34**2) / (2 * d23)
y4 = math.sqrt(d24**2 - x4**2)

ser = serial.Serial(PORT, BAUD, timeout=1)

print("UWB realtime positioning started")
print(f"A2=(0,0)  A3=({d23:.2f},0)  A4=({x4:.3f},{y4:.3f})")

while True:
    line = ser.readline().decode("utf-8", errors="ignore").strip()

    if not line.startswith("$KT1"):
        continue

    parts = line.split(",")

    try:
        # 已经实测确认：
        # A2 -> 第一个距离槽
        # A3 -> 第二个距离槽
        # A4 -> 第三个距离槽
        if parts[3] == "NULL" or parts[4] == "NULL" or parts[5] == "NULL":
            continue

        r2 = float(parts[3])
        r3 = float(parts[4])
        r4 = float(parts[5])

        x = (r2**2 - r3**2 + d23**2) / (2 * d23)

        y = (
            r2**2
            - r4**2
            + d24**2
            - 2 * x4 * x
        ) / (2 * y4)

        print(
            f"A2={r2:.2f}m  "
            f"A3={r3:.2f}m  "
            f"A4={r4:.2f}m  "
            f"--> T1 x={x:.3f}m y={y:.3f}m"
        )

    except (ValueError, IndexError):
        continue
```
